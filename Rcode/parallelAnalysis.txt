# Your data
q_matrix <- t(data)

# Function to compute eigenvalues of correlation matrix
get_eigen <- function(x) {
  eigen(cor(x))$values
}

# Observed eigenvalues
obs_eigen <- get_eigen(q_matrix)

# Number of resamples
n_iter <- 1000

# Storage
rand_eigen <- matrix(NA, nrow = n_iter, ncol = ncol(q_matrix))

set.seed(123)

for (i in 1:n_iter) {
  # Permute each column independently
  permuted <- apply(q_matrix, 2, sample, replace = FALSE)
  
  # Compute eigenvalues
  rand_eigen[i, ] <- get_eigen(permuted)
}

# Mean (or percentile) eigenvalues from resampling
mean_rand_eigen <- colMeans(rand_eigen)

# 95th percentile (recommended instead of mean)
perc95_rand_eigen <- apply(rand_eigen, 2, quantile, probs = 0.95)

# Combine results
results <- data.frame(
  Factor = 1:length(obs_eigen),
  Observed = obs_eigen,
  Mean_Random = mean_rand_eigen,
  Perc95_Random = perc95_rand_eigen
)
n_show <- min(8, nrow(results))

pretty_results <- results[1:n_show, ]
pretty_results[, -1] <- round(pretty_results[, -1], 3)

# Transpose (use Factor as column names)
t_results <- t(pretty_results[, -1])
colnames(t_results) <- paste0("F", pretty_results$Factor)
